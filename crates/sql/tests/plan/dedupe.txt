# Dedupe: ROW_NUMBER() OVER (PARTITION BY keys ORDER BY event time) = 1
# keeps a row for each key. Ascending keeps the first, which the output
# appends; descending keeps the latest, as upserts keyed by the partition
# keys. Other uses of ROW_NUMBER() are refused.

stream orders key(id) time(created_at)
  id int64 not null
  customer int64
  region utf8
  amount decimal(12, 2)
  created_at timestamp

stream events time(at)
  event_id utf8
  kind utf8
  at timestamp_ns

stream cdc key(id) changelog time(at)
  id int64 not null
  v int64
  at timestamp

# The first row of each key, with QUALIFY.
plan
INSERT INTO first_events
SELECT event_id, kind, at FROM events
QUALIFY row_number() OVER (PARTITION BY event_id ORDER BY at) = 1
----

  inputs events
  key event_id
  schema event_id Utf8, kind Utf8, at Timestamp(ns, "UTC")

  project event_id, kind, at
  need event_id
  passes event_id -> event_id, kind -> kind, at -> at
  operator dedupe-f2110852e6cff603

# The latest, read from a subquery whose number is 1: the output's key is
# the partition keys, renamed or not, and the number reads as 1. A filter
# beside it runs on the rows kept.
plan
INSERT INTO latest
SELECT customer AS c, region, amount, rn FROM (
  SELECT *, ROW_NUMBER() OVER (PARTITION BY region, customer ORDER BY created_at DESC) AS rn
  FROM orders WHERE amount > 0
) AS t
WHERE rn = 1 AND amount < 1000
----
statement 1:1 latest over
  inputs orders
  key region, c
  schema c Int64, region Utf8, amount Decimal128(12, 2), rn Int64 not null
  reads orders: customer, region, amount, created_at

  dedupe Latest by created_at on customer, region, upserts
  project customer, region, amount, rn := UInt64(1)
  filter t.amount < Decimal128(1000.00,12,2)

  passes customer -> c, region -> region, amount -> amount
