# Column names: an unquoted name matches a column without regard to case
# when exactly one column fits; a quoted name matches exactly. A column that
# passes through keeps its own spelling; a name given with AS folds to lower
# case unless quoted.

stream HR.EMPLOYEES key(EMPLOYEE_ID) changelog
  EMPLOYEE_ID int64 not null
  LAST_NAME utf8
  Hire_Date timestamp

stream mixed
  id int64
  ID utf8
  Name utf8
