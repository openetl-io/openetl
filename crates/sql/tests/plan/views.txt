# Views name a result later statements read. They're inlined into each
# reader, so keys pass through them by the same rule.

stream customers key(id)
  id int64 not null
  email utf8
  country utf8

stream orders key(id) changelog
  id int64 not null
  customer_id int64
  status utf8

plan
