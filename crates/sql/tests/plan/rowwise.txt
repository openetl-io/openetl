# Row-wise statements: projections, renames, functions, filters, UNNEST, and
# the keys they keep or drop.

stream customers key(id)
  id int64 not null
  email utf8
  name utf8
  created_at timestamp

stream orders key(id)
  id int64 not null
  region utf8
  amount decimal(12, 2)
  created_at timestamp

stream events
  user_id int64
  tags list<utf8>
  scores list<int64>

# The first statement of the docs' example replaces customers.
plan
INSERT INTO customers
SELECT id, lower(email) AS email, 'eu' AS region, created_at
FROM customers
----

  reads customers: id, email, created_at
  project id, email := lower(customers.email), region := Utf8("eu"), created_at

INSERT INTO labelled
SELECT id,
  CASE WHEN amount >= 100 THEN 'big' ELSE 'small' END AS size,
  CAST(amount AS DOUBLE) AS amount_f,
  coalesce(region, 'none') AS region
FROM orders
----
statement 1:1 labelled row_wise
  inputs orders
  key id
  schema id Int64 not null, size Utf8 not null, amount_f Float64, region Utf8 not null
  reads orders: id, region, amount
  project id, size := CASE WHEN orders.amount >= Decimal128(100.00,12,2) THEN Utf8("big") ELSE Utf8("small") END, amount_f := CAST(orders.amount AS Float64), region := CASE WHEN orders.region IS NOT NULL THEN orders.region ELSE Utf8("none") END

INSERT INTO hashed
SELECT id, md5(email) AS h, sha256(email) AS s, regexp_replace(email, '@.*', '') AS local,
  date_trunc('minute', created_at) AS minute, EXTRACT(year FROM created_at) AS year,
  substring(name FROM 1 FOR 3) AS initials, position('@' IN email) AS at_sign,
  trim(both ' ' FROM name) AS trimmed, upper(name) || '!' AS shout
FROM customers
----
statement 1:1 hashed row_wise
  inputs customers
  key id
  schema id Int64 not null, h Utf8, s Binary, local Utf8, minute Timestamp(µs, "UTC"), year Int32, initials Utf8, at_sign Int32, trimmed Utf8, shout Utf8
  reads customers: id, email, name, created_at
  project id, h := CAST(md5(customers.email) AS Utf8), s := sha256(customers.email), local := regexp_replace(customers.email, Utf8("@.*"), Utf8("")), minute := date_trunc(Utf8("minute"), customers.created_at), year := date_part(Utf8("YEAR"), customers.created_at), initials := substr(customers.name, Int64(1), Int64(3)), at_sign := strpos(customers.email, Utf8("@")), trimmed := btrim(customers.name, Utf8(" ")), shout := upper(customers.name) || Utf8("!")

# A derived table is read like a view.
plan
INSERT INTO active
SELECT customer, email FROM (SELECT id AS customer, lower(email) AS email FROM customers WHERE id > 1) AS c
WHERE email LIKE '%@x.io'
----
statement 1:1 active row_wise
  inputs customers
  key customer
  schema customer Int64 not null, email Utf8
  reads customers: id, email
  filter customers.id > Int64(1)
  project customer := customers.id, email := lower(customers.email)
  filter c.email LIKE Utf8("%@x.io")

# Once aliased, the stream's own name doesn't qualify its columns, as in
# Postgres.
plan
INSERT INTO q SELECT customers.id FROM customers AS c
----
error bind 1:22 there's no column customers.id; there's c.id
