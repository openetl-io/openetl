# What's refused, each where it's written: later tiers' statements,
# functions that aren't deterministic in keys, a changelog's key, and SQL
# transforms don't run.

stream orders key(id)
  id int64 not null
  region utf8
  amount decimal(12, 2)
  created_at timestamp

stream customers key(id)
  id int64 not null
  email utf8

stream cdc key(id) changelog
  id int64 not null
  name utf8
  tags list<utf8>

stream cdc2 key(tenant, id) changelog
  tenant utf8 not null
  id int64 not null
  name utf8

# A changelog stream's key passes through every statement unchanged,
# renamed or not.

----

  key person
  schema person Int64 not null, name Utf8
