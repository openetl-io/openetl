# JSON paths: our own functions under datafusion-functions-json's names, and
# its operators. They read JSON columns and plain strings alike. Cases with
# a literal document are folded when planned, so they show what a call gives;
# tests/eval.rs evaluates the others on batches.

stream events key(id)
  id int64 not null
  doc json
  raw utf8
  field utf8
  pos int64
  day date

# ->> is json_as_text, ? is json_contains, and a chain of -> is one call.
plan
INSERT INTO x
SELECT id, doc->>'name' AS name, doc->'user'->>'email' AS email, doc->'items'->0->>'sku' AS first_sku
FROM events
WHERE doc->>'status' = 'active' AND doc ? 'user'
----
statement 1:1 x row_wise
  inputs events
  key id
  schema id Int64 not null, name Utf8, email Utf8, first_sku Utf8
  reads events: id, doc
  filter json_as_text(events.doc, Utf8("status")) = Utf8("active") AND json_contains(events.doc, Utf8("user"))
  project id, name := json_as_text(events.doc, Utf8("name")), email := json_as_text(events.doc, Utf8("user"), Utf8("email")), first_sku := json_as_text(events.doc, Utf8("items"), Int64(0), Utf8("sku"))

# -> is json_get, whose value can be of any JSON type. An output column holds
# it as JSON text, exactly as the document writes it.
plan
INSERT INTO x SELECT id, doc->'user' AS "user", json_get(raw, 'a', -1) AS last_a FROM events
----
statement 1:1 x row_wise
  inputs events
  key id
  schema id Int64 not null, user Utf8 (json), last_a Utf8 (json)
  reads events: id, doc, raw
  project id, user := json_union_to_text(events.doc, Utf8("user")), last_a := json_union_to_text(events.raw, Utf8("a"), Int64(-1))

# A cast of json_get's value reads it with the getter of that type, so a
# value of another type is null; a cast to a string type is ->>'s text.
plan
