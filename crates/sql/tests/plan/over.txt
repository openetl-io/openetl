# OVER: sum, count, min, max, avg, lag and first_value over a frame of each
# key's rows that ends at the current row, by event time, as one operator
# with a need on its partition keys.

stream orders key(id) time(created_at)
  id int64 not null
  customer int64
  region utf8
  amount decimal(12, 2)
  qty int32
  tags list<utf8>
  created_at timestamp

stream ticks time(at)
  sym utf8
  price float64
  at timestamp_ns

stream cdc key(id) changelog time(at)
  id int64 not null
  v int64
  at timestamp

stream plain key(id)
  id int64 not null
  amount int64
  at timestamp
